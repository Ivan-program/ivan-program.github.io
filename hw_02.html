WEB.HW.02.01

Задание

Напишите функцию, которая принимает три массива и находит сумму всех элементов этих массивов.

Решение

const arr1 = [1, 2, 3];
const arr2 = [4, 5, 6];
const arr3 = [7, 8, 9];

function sumOfAllArrays(arr1, arr2, arr3) {
    const combo = [...arr1, ...arr2, ...arr3];
    let sum = 0;
    for (let i = 0; i < combo.length; i++) {
        sum += combo[i];
    }
    return sum;
}

console.log(sumOfAllArrays(arr1, arr2, arr3));

WEB.HW.02.02

Задание

Напишите функцию, которая принимает три массива и возвращает сумму всех простых чисел, которые есть в этих массивах.

Решение

const nums1 = [2, 3, 4, 5, 6];
const nums2 = [7, 8, 9, 10, 11];
const nums3 = [12, 13, 14, 15, 16];

function isPrime(n) {
    if (n < 2) {
        return false;
    }
    for (let i = 2; i * i <= n; i++) {
        if (n % i === 0) {
            return false;
        }
    }
    return true;
}

function sumOfPrimesInArrays(arr1, arr2, arr3) {
    const allNumbers = [...arr1, ...arr2, ...arr3];
    let sum = 0;
    for (let i = 0; i < allNumbers.length; i++) {
        if (isPrime(allNumbers[i])) {
            sum += allNumbers[i];
        }
    }
    return sum;
}

console.log(sumOfPrimesInArrays(nums1, nums2, nums3));

function formatValue(value) {
    if (Array.isArray(value)) {
        return value.join(", ");
    }
    if (typeof value === "object" && value !== null) {
        let parts = [];
        for (let key in value) {
            parts.push(key + ": " + value[key]);
        }
        return parts.join(", ");
    }
    return value;
}

function commonProps(...objs) {
    if (objs.length === 0) {
        return;
    }

    for (let key in objs[0]) {
        let inAll = true;
        for (let i = 1; i < objs.length; i++) {
            if (!(key in objs[i])) {
                inAll = false;
                break;
            }
        }

        if (inAll) {
            let values = [];
            for (let i = 0; i < objs.length; i++) {
                values.push(formatValue(objs[i][key]));
            }
            console.log(key + ": " + values.join(", "));
        }
    }
}

commonProps(shop1, shop2, shop3);
